#include <iostream>
using namespace std;

int main(){
    char combu;
    float quant, total;
    float gas = 5.80, eta = 4.20;
    
    cout << "Tipo de combustível: (G = Gasolina | E = Etanol)" << endl;
    cin >> combu;
    cout << "Digite a quantidade desejada: " << endl;
    cin >> quant;
    
    
    
    if(combu == 'G' || combu == 'g'){
        total = quant * gas;    
    } else if  (combu == 'E' || combu == 'e'){
        total = quant * eta;
    } else {
        cout << "Opção inválida!";
    };
    
    cout << "O total a pagar foi: R$" << total << endl;
    if(total > 150){
        cout << "Seu valor ultrapassou R$150,00, Você ganhou um desconto especial na lavagem"
    }
}
